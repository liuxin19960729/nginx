// ngx_list_part_t 链表的一个元素
typedef struct ngx_list_part_s  ngx_list_part_t;

// 链表元素 这样设计非常灵活 可以是任何一种数据结构
struct ngx_list_part_s {
    void             *elts;
    ngx_uint_t        nelts;
    ngx_list_part_t  *next;
};

// ngx_list_t 链表容器
typedef struct {
    ngx_list_part_t  *last;
    ngx_list_part_t   part;
    size_t            size;
    ngx_uint_t        nalloc;
    ngx_pool_t       *pool;
} ngx_list_t;
